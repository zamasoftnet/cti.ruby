478 def write(str)
479   e = str.encoding;
480   str.force_encoding(Encoding::ASCII_8BIT)
481   # SSLSocket は write_nonblock を持つが nonblock= を持たない。
482   # 両方あるときだけ非ブロック書き込みを使う(2026-09-10。
483   # ctips:// で `undefined method 'nonblock='` になっていた)
484   bio = @io.respond_to?('write_nonblock') && @io.respond_to?('nonblock=')
485   while str.bytesize > 0
486     data = str.slice!(0, CTIP2::CTI_BUFFER_SIZE)
487     packet = [data.bytesize + 1, CTIP2::REQ_DATA].pack('NC') + data
488     while packet.bytesize > 0
489       rs, ws = IO::select([@io], [@io])
490       if packet.bytesize > 0 && ws[0]
491         if bio
492           len = @io.write_nonblock(packet)
493           @io.nonblock = false
494         else
495           len = @io.write(packet)
496         end
497         packet.slice!(0, len)
498       end
499       if rs[0]
500         @session.build_next
501       end
502     end
503   end
504   str.force_encoding(e)
